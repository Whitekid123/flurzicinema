import {describe,it,expect,vi} from 'vitest';
import {acceptSnapshot,allReady,canControl,createRoomController,invitationCode,isHost,playbackPosition,roomDeadline,roomError,validRoomName,validSnapshot,type RoomSnapshot,type RoomService} from '../lib/rooms';
import {createRoomPreview} from '../lib/api/mocks/room-preview';
import {unavailableRoomService} from '../lib/api/room-service';
import {createMockServices} from '../lib/api/mocks/mockServices';

const snapshot=(patch:Partial<RoomSnapshot>={}):RoomSnapshot=>({id:'room',name:'Movie night',revision:1,phase:'lobby',media:{titleId:'film'},viewerProfileId:'p',selfId:'self',participants:[{id:'self',profileId:'p',name:'Viewer',role:'host',presence:'joined',ready:false}],privacy:'invite-only',controls:'host',playback:{titleId:'film',playing:false,positionSeconds:0,serverTimeMs:1000},serverClockOffsetMs:0,isPreview:false,...patch});
async function preview(){const services=createMockServices();const session=await services.auth.login({email:'demo@flurzi.example',password:'preview-only'});const profile=(await services.profiles.listProfiles())[0];const rooms=createRoomPreview(services.auth,services.profiles,services.catalog,profile,session.session.id);const title=(await services.catalog.listTitles()).items.find(t=>t.mediaType==='movie')!;return {services,session,profile,rooms,title};}
async function create(rooms:RoomService,titleId:string){return rooms.create({name:'Our evening',media:{titleId},privacy:'invite-only',controls:'host'});}
describe('room permissions and readiness',()=>{
 it('identifies a service-confirmed host for this viewer',()=>{expect(isHost(snapshot())).toBe(true);expect(isHost(snapshot({viewerProfileId:'other'}))).toBe(false);});
 it('does not give a participant host playback controls',()=>{const room=snapshot({phase:'viewing'});room.participants[0].role='participant';expect(canControl(room)).toBe(false);expect(canControl({...room,controls:'everyone'})).toBe(true);});
 it('does not expose controls to a left viewer',()=>{const room=snapshot({phase:'viewing'});room.participants[0].presence='left';expect(isHost(room)).toBe(false);expect(canControl(room)).toBe(false);});
 it('requires all joined participants to be ready and connected',()=>{const room=snapshot();expect(allReady(room)).toBe(false);room.participants[0].ready=true;expect(allReady(room)).toBe(true);room.participants[0].presence='disconnected';expect(allReady(room)).toBe(false);});
 it('excludes pending invitations and departed participants from readiness',()=>{const room=snapshot();room.participants[0].ready=true;room.participants.push({id:'invited',profileId:'x',name:'Invited',role:'participant',presence:'invited',ready:false});expect(allReady(room)).toBe(true);});
 it('does not start an empty room',()=>expect(allReady(snapshot({participants:[]}))).toBe(false));
 it('validates trimmed names',()=>{expect(validRoomName(' Friday ')).toBe(true);expect(validRoomName('  ')).toBe(false);expect(validRoomName('x'.repeat(81))).toBe(false);});
});
describe('authoritative snapshots and synchronization',()=>{
 it('rejects mismatched viewer and room identities',()=>{expect(()=>validSnapshot(snapshot(),'other')).toThrow();expect(()=>validSnapshot(snapshot(),'p','other')).toThrow();});
 it('rejects inconsistent playback media and invalid positions',()=>{expect(()=>validSnapshot(snapshot({playback:{titleId:'other',playing:true,positionSeconds:0,serverTimeMs:0}}),'p')).toThrow();expect(()=>validSnapshot(snapshot({playback:{titleId:'film',playing:true,positionSeconds:NaN,serverTimeMs:0}}),'p')).toThrow();});
 it('rejects negative and fractional revisions',()=>{for(const revision of [-1,1.5,NaN])expect(()=>validSnapshot(snapshot({revision}),'p')).toThrow();});
 it('ignores stale and duplicate snapshots',()=>{const a=snapshot({revision:4});expect(acceptSnapshot(a,snapshot({revision:3}),'p')).toBe(a);expect(acceptSnapshot(a,snapshot({revision:4}),'p')).toBe(a);});
 it('accepts ordered transitions and host changes',()=>{const a=snapshot();const next=snapshot({revision:2,phase:'viewing'});next.participants[0].role='participant';expect(acceptSnapshot(a,next,'p')).toBe(next);expect(isHost(next)).toBe(false);});
 it('does not regress viewing to lobby or revive ended rooms',()=>{const a=snapshot({phase:'viewing'});expect(acceptSnapshot(a,snapshot({revision:2}),'p')).toBe(a);const ended=snapshot({phase:'ended'});expect(acceptSnapshot(ended,snapshot({revision:9}),'p')).toBe(ended);});
 it('requires a new controller to change media identity',()=>{const a=snapshot();const next=snapshot({revision:2,media:{titleId:'other'},playback:{titleId:'other',playing:false,positionSeconds:0,serverTimeMs:0}});expect(acceptSnapshot(a,next,'p')).toBe(a);});
 it('projects server play position and keeps pause still',()=>{const p=snapshot().playback;expect(playbackPosition({...p,playing:true},5000)).toBe(4);expect(playbackPosition(p,5000)).toBe(0);expect(playbackPosition({...p,playing:true},0)).toBe(0);});
 it('disconnects and requires a fresh authorized resync',()=>{const c=createRoomController('p','room');c.resync(snapshot({revision:4}));c.disconnect();expect(c.connected()).toBe(false);expect(()=>c.resync(snapshot({revision:2}))).toThrow();expect(c.resync(snapshot({revision:4})).revision).toBe(4);expect(c.connected()).toBe(true);});
 it('deduplicates confirmed reactions and rejects unknown senders',()=>{const c=createRoomController('p','room');c.resync(snapshot());const event={type:'reaction',roomId:'room',participantId:'self',reaction:'heart',id:'event',revision:1} as const;expect(c.receive(event)).toBe(true);expect(c.receive(event)).toBe(false);expect(c.receive({...event,id:'2',participantId:'ghost'})).toBe(false);});
 it('rejects old, unrelated and disconnected reactions',()=>{const c=createRoomController('p','room');c.resync(snapshot());const event={type:'reaction',roomId:'room',participantId:'self',reaction:'heart',id:'event',revision:0} as const;expect(c.receive(event)).toBe(false);expect(c.receive({...event,revision:1,roomId:'other'})).toBe(false);c.disconnect();expect(c.receive({...event,revision:1})).toBe(false);});
 it('bounds requests that never settle',async()=>{await expect(roomDeadline(new Promise(()=>{}),5)).rejects.toMatchObject({code:'SYNC_UNAVAILABLE'});await expect(roomDeadline(Promise.resolve('ok'),100)).resolves.toBe('ok');});
});
describe('invitation syntax does not authorize playback',()=>{
 it('accepts a code or same-origin join link',()=>{expect(invitationCode(' abcdef_123 ','https://cinema.test')).toBe('abcdef_123');expect(invitationCode('https://cinema.test/browse/rooms/join?code=abcdef','https://cinema.test')).toBe('abcdef');});
 it.each(['https://evil.test/browse/rooms/join?code=abcdef','javascript:alert(1)','https://cinema.test/video.mp4','abc','//evil.test/a','https://cinema.test/browse/rooms/join?code=abcdef#x'])('rejects unsafe or malformed input %s',value=>expect(()=>invitationCode(value,'https://cinema.test')).toThrow());
 it('maps supported service errors without exposing details',()=>{for(const code of ['INVITATION_EXPIRED','ROOM_FULL','ROOM_ENDED','ACCESS_DENIED','SESSION_EXPIRED','SUBSCRIPTION_REQUIRED','REGION_RESTRICTED','RATE_LIMITED'])expect(roomError({code,message:'secret'})).not.toContain('secret');});
 it('production adapter stays unavailable',async()=>{expect((await unavailableRoomService.configuration()).available).toBe(false);await expect(unavailableRoomService.create({} as never)).rejects.toMatchObject({code:'SERVICE_UNAVAILABLE'});});
});
describe('isolated authenticated development preview',()=>{
 it('creates a lobby from a real catalog identity with no phantom participants',async()=>{const {rooms,title,services}=await preview();const room=await create(rooms,title.id);expect(room.media.titleId).toBe(title.id);expect(room.phase).toBe('lobby');expect(room.isPreview).toBe(true);expect(room.participants).toHaveLength(1);expect((await services.subscriptions.getEntitlement()).canStream).toBe(false);});
 it('validates metadata identity without inventing a media source',async()=>{const {rooms}=await preview();await expect(create(rooms,'https://evil.test/movie.mp4')).rejects.toMatchObject({code:'VALIDATION'});});
 it('runs ready, start, playback intents, reconnect and end transitions',async()=>{const {rooms,title}=await preview();let room=await create(rooms,title.id);await expect(rooms.start(room.id)).rejects.toMatchObject({code:'NOT_READY'});room=await rooms.setReady(room.id,true);expect(allReady(room)).toBe(true);room=await rooms.start(room.id);expect(room.phase).toBe('viewing');room=await rooms.sendPlaybackIntent(room.id,{action:'play',positionSeconds:5});expect(room.playback.playing).toBe(true);expect((await rooms.reconnect(room.id)).revision).toBe(room.revision);room=await rooms.end(room.id);expect(room.phase).toBe('ended');await expect(rooms.setReady(room.id,true)).rejects.toMatchObject({code:'ROOM_ENDED'});});
 it('does not resolve arbitrary codes and rejects ended invitations',async()=>{const {rooms,title}=await preview();await expect(rooms.resolveInvitation('invented')).rejects.toMatchObject({code:'INVITATION_INVALID'});const room=await create(rooms,title.id);expect((await rooms.resolveInvitation(room.invitation!.code)).id).toBe(room.id);expect((await rooms.join(room.invitation!.code)).id).toBe(room.id);await rooms.end(room.id);await expect(rooms.resolveInvitation(room.invitation!.code)).rejects.toMatchObject({code:'ROOM_ENDED'});});
 it('only emits confirmed self reactions and applies rate limits',async()=>{const {rooms,title}=await preview();const room=await create(rooms,title.id);await rooms.setReady(room.id,true);await rooms.start(room.id);const incoming=vi.fn();const stop=rooms.subscribe(room.id,incoming,()=>{});await rooms.sendReaction(room.id,'heart');expect(incoming.mock.calls.some(([e])=>e.type==='reaction'&&e.participantId===room.selfId)).toBe(true);await expect(rooms.sendReaction(room.id,'heart')).rejects.toMatchObject({code:'RATE_LIMITED'});stop();});
 it('isolates profiles and session identities',async()=>{const {rooms,title,services,profile,session}=await preview();const room=await create(rooms,title.id);const other=(await services.profiles.listProfiles()).find(p=>p.id!==profile.id)!;const otherRooms=createRoomPreview(services.auth,services.profiles,services.catalog,other,session.session.id);await expect(otherRooms.getSnapshot(room.id)).rejects.toMatchObject({code:'NOT_FOUND'});await expect(otherRooms.resolveInvitation(room.invitation!.code)).rejects.toMatchObject({code:'INVITATION_INVALID'});const differentSession=createRoomPreview(services.auth,services.profiles,services.catalog,profile,'other-session');await expect(differentSession.list()).rejects.toMatchObject({code:'SESSION_EXPIRED'});});
 it('rejects reads and mutations after logout',async()=>{const {rooms,title,services}=await preview();const room=await create(rooms,title.id);await services.auth.logout();await expect(rooms.getSnapshot(room.id)).rejects.toMatchObject({code:'SESSION_EXPIRED'});await expect(rooms.setReady(room.id,true)).rejects.toMatchObject({code:'SESSION_EXPIRED'});});
 it('expires invitations using service time',async()=>{const {rooms,title}=await preview();const room=await create(rooms,title.id);const now=Date.now();const clock=vi.spyOn(Date,'now').mockReturnValue(now+3600001);try{await expect(rooms.resolveInvitation(room.invitation!.code)).rejects.toMatchObject({code:'INVITATION_EXPIRED'});}finally{clock.mockRestore();}});
 it('does not share rooms between independent auth adapters',async()=>{const first=await preview();const room=await create(first.rooms,first.title.id);const second=await preview();await expect(second.rooms.getSnapshot(room.id)).rejects.toMatchObject({code:'NOT_FOUND'});});
 it('host departure closes the room',async()=>{const {rooms,title}=await preview();const room=await create(rooms,title.id);await rooms.leave(room.id);expect((await rooms.getSnapshot(room.id)).phase).toBe('ended');expect((await rooms.list()).current.some(r=>r.id===room.id)).toBe(false);});
});
