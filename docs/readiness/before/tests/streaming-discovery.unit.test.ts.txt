describe("catalog queries and URL state",()=>{
 it("normalizes invalid URL values and limits search size",()=>{const state=readDiscoveryState(new URLSearchParams("q="+ "a".repeat(180)+"&page=-1&type=anything&sort=popular&year=abc"));expect(state).toEqual({q:"a".repeat(150),genre:"",page:1,sort:"year-desc",type:undefined,year:undefined});});
 it("round-trips shareable filters, Unicode queries and pagination",()=>{const state=readDiscoveryState(new URLSearchParams("q=Quiet+%26+Beyond&genre=mystery&year=2026&type=movie&sort=title-asc&page=2"));const href=discoveryHref("/browse/search",state);expect(readDiscoveryState(new URLSearchParams(href.split("?")[1]))).toEqual(state);});
 it("omits default URL state and clamps excessive page values",()=>{expect(discoveryHref("/browse/movies",readDiscoveryState(new URLSearchParams()))).toBe("/browse/movies");expect(readDiscoveryState(new URLSearchParams("page=99999999999")).page).toBe(10000);expect(readDiscoveryState(new URLSearchParams("page=1.5")).page).toBe(1);});
 it("passes combined filters to the server request with fixed page size",()=>{const state=readDiscoveryState(new URLSearchParams("q=Quiet&genre=mystery&year=2026&type=series&page=3"));expect(discoveryRequest(state,"movie")).toEqual({page:3,pageSize:24,search:"Quiet",genre:"mystery",year:2026,sort:"year-desc",mediaType:"movie"});});
 it("separates movies from series using domain mediaType",()=>{expect(queryCatalog(mockTitles,{mediaType:"movie"}).items).toHaveLength(4);expect(queryCatalog(mockTitles,{mediaType:"series"}).items.map(t=>t.title)).toEqual(["Velvet Hours"]);});
 it("combines genre, release year and case-insensitive title/synopsis search",()=>{expect(queryCatalog(mockTitles,{mediaType:"movie",genre:"mystery",year:2026,search:"QUIET"}).items.map(t=>t.id)).toEqual(["mock-quiet-beyond"]);expect(queryCatalog(mockTitles,{search:"oceanographer"}).items[0].id).toBe("mock-tides");expect(queryCatalog(mockTitles,{genre:"unknown"}).total).toBe(0);});
 it("sorts on actual title and release-year fields without mutating records",()=>{const before=mockTitles.map(t=>t.id);expect(queryCatalog(mockTitles,{sort:"year-asc"}).items[0].title).toBe("Afterlight");expect(queryCatalog(mockTitles,{sort:"year-desc"}).items[0].title).toBe("The Quiet Beyond");expect(queryCatalog(mockTitles,{sort:"title-asc"}).items[0].title).toBe("Afterlight");expect(mockTitles.map(t=>t.id)).toEqual(before);});
 it("paginates after filtering and preserves total result count",()=>{const a=queryCatalog(mockTitles,{page:1,pageSize:2});const b=queryCatalog(mockTitles,{page:2,pageSize:2});expect(a.total).toBe(5);expect(b.items).toHaveLength(2);expect(a.items.some(t=>b.items.some(i=>i.id===t.id))).toBe(false);expect(queryCatalog(mockTitles,{page:99,pageSize:2}).items).toEqual([]);});
 it("puts unknown release years after known years",()=>{const titles=[{...mockTitles[0],id:"unknown",releaseYear:undefined},mockTitles[4]];expect(queryCatalog(titles,{sort:"year-asc"}).items[1].id).toBe("unknown");expect(queryCatalog(titles,{sort:"year-desc"}).items[1].id).toBe("unknown");});
 it("development service respects paged search and query filters",async()=>{const s=createMockServices();const results=await s.catalog.search("a",{page:2,pageSize:1,sort:"title-asc"});expect(results.page).toBe(2);expect(results.pageSize).toBe(1);expect(results.items).toHaveLength(1);expect((await s.catalog.listTitles({mediaType:"series",genre:"drama"})).items[0].title).toBe("Velvet Hours");});
});
describe("title metadata and history boundaries",()=>{
 it("resolves unique title IDs and distinguishes missing records from failure",async()=>{const getTitle=vi.fn(async(id:string)=>{if(id==="failure")throw Error();return id==="missing"?null:mockTitles[0];});const rows=await resolveTitleRecords(["mock-quiet-beyond","mock-quiet-beyond","missing","failure"],{getTitle} as unknown as CatalogService);expect(getTitle).toHaveBeenCalledTimes(3);expect(rows.map(r=>[r.id,r.title?.id,r.failed])).toEqual([["mock-quiet-beyond","mock-quiet-beyond",false],["missing",undefined,false],["failure",undefined,true]]);});
 it("rejects mismatched title IDs rather than showing unrelated artwork",async()=>{const rows=await resolveTitleRecords(["wrong"],{getTitle:async()=>mockTitles[0]} as unknown as CatalogService);expect(rows).toEqual([{id:"wrong",title:null,failed:true}]);});
 it("reuses route-local metadata and shares pending requests",async()=>{const getTitle=vi.fn(async()=>mockTitles[0]);const resolver=createTitleResolver({getTitle} as unknown as CatalogService);await Promise.all([resolver.resolve([mockTitles[0].id]),resolver.resolve([mockTitles[0].id])]);await resolver.resolve([mockTitles[0].id]);expect(getTitle).toHaveBeenCalledTimes(1);});
 it("retries failed metadata instead of caching a service failure",async()=>{const getTitle=vi.fn().mockRejectedValueOnce(Error()).mockResolvedValue(mockTitles[0]);const resolver=createTitleResolver({getTitle} as unknown as CatalogService);expect((await resolver.resolve([mockTitles[0].id]))[0].failed).toBe(true);expect((await resolver.resolve([mockTitles[0].id]))[0].title?.id).toBe(mockTitles[0].id);expect(getTitle).toHaveBeenCalledTimes(2);});
 it("bounds metadata request concurrency to eight",async()=>{let active=0,max=0;const getTitle=async()=>{active++;max=Math.max(active,max);await new Promise(r=>setTimeout(r,1));active--;return null;};await resolveTitleRecords(Array.from({length:25},(_,i)=>String(i)),{getTitle} as unknown as CatalogService);expect(max).toBeLessThanOrEqual(8);});
 it("keeps primary title data when optional services fail",async()=>{const s=createMockServices();s.subscriptions.getEntitlement=vi.fn().mockRejectedValue(Error());s.catalog.listRecommendations=vi.fn().mockRejectedValue(Error());const detail=await loadTitleDetails(s,mockTitles[0].id);expect(detail?.title.id).toBe(mockTitles[0].id);expect(detail?.related).toEqual([]);expect(detail?.issues).toEqual(["related titles","account viewing access"]);});
 it("does not fetch related metadata for unknown titles",async()=>{const s=createMockServices();s.catalog.listRecommendations=vi.fn();expect(await loadTitleDetails(s,"unknown")).toBeNull();expect(s.catalog.listRecommendations).not.toHaveBeenCalled();});
 it("deduplicates related titles and excludes the current title",async()=>{const s=createMockServices();s.catalog.listRecommendations=vi.fn().mockResolvedValue([mockTitles[0],mockTitles[4],mockTitles[4]]);expect((await loadTitleDetails(s,mockTitles[0].id))?.related).toEqual([mockTitles[4]]);});
 it("selects actual seasons by ID with a safe fallback",()=>{const series=mockTitles.find(t=>t.mediaType==="series")!;if(series.mediaType!=="series")throw Error();const second={...series.seasons[0],id:"season-two",number:2};const record={...series,seasons:[...series.seasons,second]};expect(selectedSeason(record,"season-two")?.number).toBe(2);expect(selectedSeason(record,"invalid")?.number).toBe(1);expect(selectedSeason({...record,seasons:[]},null)).toBeNull();});
 it("preserves five fictional title/artwork relationships and non-playable status",()=>{expect(mockTitles).toHaveLength(5);for(const t of mockTitles){expect(t.artwork.posterUrl).toBe("/images/"+t.id.replace("mock-","")+".webp");expect(t.availability.status).toBe("metadata-only");expect(t.isLicensedForPlayback).toBe(false);}expect((mockTitles.find(t=>t.mediaType==="series") as Extract<CatalogTitle,{mediaType:"series"}>).seasons[0].episodes).toHaveLength(2);});
 it("does not fabricate history and validates profile ownership",async()=>{const s=createMockServices();await expect(s.catalog.getViewingHistory("mock-profile-main")).rejects.toThrow();await s.auth.login({email:"demo@flurzi.example",password:"fixture"});const history=await s.catalog.getViewingHistory("mock-profile-main",{pageSize:12});expect(history).toEqual({items:[],page:1,pageSize:12,total:0});await expect(s.catalog.getViewingHistory("unknown")).rejects.toThrow();});
 it("labels mock metadata honestly and blocks expired/invalid availability claims",()=>{expect(availabilityCopy({status:"licensed"},true,true)).toContain("Fictional");expect(availabilityCopy({status:"metadata-only"},false,false)).toContain("not currently available");expect(availabilityCopy({status:"licensed",availableUntil:"2000-01-01"},true,false)).toContain("not currently");expect(availabilityCopy({status:"licensed",availableFrom:"invalid"},true,false)).toContain("not currently");expect(availabilityCopy({status:"licensed"},true,false)).toContain("fresh server authorization");});
});

 let records:WatchlistItem[]=[];
 const service:WatchlistService={
  listWatchlist:vi.fn(async(profileId)=>structuredClone(records.filter(r=>r.profileId===profileId))),
  addToWatchlist:vi.fn(async(profileId,titleId)=>{const item={id:profileId+titleId,profileId,titleId,addedAt:"2026-01-01"};records.push(item);return item;}),
  removeFromWatchlist:vi.fn(async(profileId,titleId)=>{records=records.filter(r=>r.profileId!==profileId||r.titleId!==titleId);}),

describe("profile watchlist controller",()=>{
 it("synchronizes mounted views without leaking another profile's data",async()=>{const service=fixtureWatchlist();const a=createWatchlistController(service,"main"),b=createWatchlistController(service,"main"),kids=createWatchlistController(service,"kids");const stop=[a.start(),b.start(),kids.start()];try{await vi.waitFor(()=>expect(a.getSnapshot().loading).toBe(false));await a.toggle("title","Title");await vi.waitFor(()=>expect(b.getSnapshot().items).toHaveLength(1));expect(kids.getSnapshot().items).toEqual([]);await b.remove("title");await vi.waitFor(()=>expect(a.getSnapshot().items).toEqual([]));}finally{stop.forEach(fn=>fn());}});
 it("keeps failed list requests distinct from empty lists",async()=>{const service=fixtureWatchlist();service.listWatchlist=vi.fn().mockRejectedValue(Error());const controller=createWatchlistController(service,"main");const stop=controller.start();try{await vi.waitFor(()=>expect(controller.getSnapshot().error).not.toBe(""));expect(controller.getSnapshot().items).toBeNull();await controller.toggle("a","A");expect(service.addToWatchlist).not.toHaveBeenCalled();}finally{stop();}});
 it("announces mutation failure without changing confirmed saved state",async()=>{const service=fixtureWatchlist();service.addToWatchlist=vi.fn().mockRejectedValue(Error());const controller=createWatchlistController(service,"main");const stop=controller.start();try{await vi.waitFor(()=>expect(controller.getSnapshot().loading).toBe(false));await controller.toggle("a","A");expect(controller.getSnapshot().items).toEqual([]);expect(controller.getSnapshot().notice).toContain("Could not update");expect(controller.getSnapshot().pending).toEqual([]);}finally{stop();}});
 it("ignores an older list response after a retry has completed",async()=>{const service=fixtureWatchlist();let resolve!:(rows:WatchlistItem[])=>void;service.listWatchlist=vi.fn().mockImplementationOnce(()=>new Promise<WatchlistItem[]>(r=>{resolve=r;})).mockResolvedValue([]);const controller=createWatchlistController(service,"main");const stop=controller.start();try{controller.retry();await vi.waitFor(()=>expect(controller.getSnapshot().loading).toBe(false));resolve([{id:"old",titleId:"stale",profileId:"main",addedAt:"old"}]);await Promise.resolve();expect(controller.getSnapshot().items).toEqual([]);}finally{stop();}});
 it("prevents duplicate in-flight mutations for the same title",async()=>{const service=fixtureWatchlist();let resolve!:(item:WatchlistItem)=>void;service.addToWatchlist=vi.fn(()=>new Promise<WatchlistItem>(r=>{resolve=r;}));const controller=createWatchlistController(service,"main");const stop=controller.start();try{await vi.waitFor(()=>expect(controller.getSnapshot().loading).toBe(false));const first=controller.toggle("a","A");await controller.toggle("a","A");expect(service.addToWatchlist).toHaveBeenCalledTimes(1);resolve({id:"saved",profileId:"main",titleId:"a",addedAt:"now"});await first;expect(controller.getSnapshot().items).toHaveLength(1);}finally{stop();}});
});
