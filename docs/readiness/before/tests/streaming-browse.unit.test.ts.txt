import {describe,expect,it,vi} from "vitest";
import {createMockServices} from "../lib/api/mocks/mockServices";
import {mockTitles} from "../lib/api/mocks/mockData";
import {episodeLabel,loadBrowseHome,playbackAllowed,progressDisplay,titleHref,uniqueTitles} from "../lib/browse-home";
import type {Entitlement,ViewingProgress} from "../types/streaming";
const entitlement:Entitlement={accountId:"a",canStream:true,canDownload:false,maxStreams:1,maxProfiles:5};
const progress:ViewingProgress={id:"p",profileId:"mock-profile-main",titleId:"mock-velvet-hours",episodeId:"mock-vh-s1e1",positionSeconds:600,durationSeconds:2400,completed:false,lastWatchedAt:"2026-01-01T00:00:00Z"};
async function signedIn(){const services=createMockServices();await services.auth.login({email:"demo@flurzi.example",password:"fixture"});return services;}
describe("streaming home logic",()=>{
 it("never permits demo playback even with an entitlement",()=>{for(const t of mockTitles)expect(playbackAllowed(t,entitlement)).toBe(false);});
 it("requires both license and entitlement for a movie",()=>{const movie={...mockTitles[0],isLicensedForPlayback:true,availability:{status:"licensed" as const}};expect(playbackAllowed(movie,null)).toBe(false);expect(playbackAllowed(movie,{...entitlement,canStream:false})).toBe(false);expect(playbackAllowed(movie,entitlement)).toBe(true);});
 it("rejects expired and future availability windows",()=>{const movie={...mockTitles[0],isLicensedForPlayback:true,availability:{status:"licensed" as const,availableUntil:"2000-01-01T00:00:00Z"}};expect(playbackAllowed(movie,entitlement)).toBe(false);expect(playbackAllowed({...movie,availability:{status:"licensed",availableFrom:"invalid-date"}},entitlement)).toBe(false);expect(playbackAllowed({...movie,availability:{status:"licensed",availableFrom:"2100-01-01T00:00:00Z"}},entitlement)).toBe(false);});
 it("requires a matching licensed episode before offering resume",()=>{const series=mockTitles.find(t=>t.mediaType==="series")!;if(series.mediaType!=="series")throw Error();const title={...series,isLicensedForPlayback:true,availability:{status:"licensed" as const}};expect(playbackAllowed(title,entitlement,progress)).toBe(false);expect(playbackAllowed({...title,seasons:title.seasons.map(s=>({...s,episodes:s.episodes.map(e=>({...e,isLicensedForPlayback:true,availability:{status:"licensed" as const}}))}))},entitlement,progress)).toBe(true);expect(playbackAllowed(title,entitlement)).toBe(false);});
 it("clamps progress and handles unknown duration",()=>{expect(progressDisplay({...progress,durationSeconds:Infinity})).toEqual({percent:0,remaining:null});expect(progressDisplay(progress)).toEqual({percent:25,remaining:30});expect(progressDisplay({...progress,positionSeconds:9000})).toEqual({percent:100,remaining:null});expect(progressDisplay({...progress,positionSeconds:-1,durationSeconds:undefined})).toEqual({percent:0,remaining:null});});
 it("matches season and episode metadata without inventing it",()=>{const series=mockTitles.find(t=>t.mediaType==="series")!;expect(episodeLabel(series,progress)).toBe("S1 · E1 — An unopened letter");expect(episodeLabel(series,{...progress,episodeId:"unknown"})).toBeNull();});
 it("deduplicates title records and encodes detail routes",()=>{expect(uniqueTitles([...mockTitles,...mockTitles])).toHaveLength(5);expect(titleHref({...mockTitles[0],id:"a/b"})).toBe("/browse/title/a%2Fb");});
 it("loads service collections and leaves unwatched profiles empty",async()=>{const s=await signedIn();const home=await loadBrowseHome(s,"mock-profile-main");expect(home.titles).toHaveLength(5);expect(home.progress).toEqual([]);expect(home.saved).toEqual([]);expect(home.issues).toEqual([]);expect(home.entitlement?.canStream).toBe(false);expect(home.rails.map(r=>r.title)).toEqual(["Movies to explore","Series spotlight"]);});
 it("keeps useful catalog data when optional services fail",async()=>{const s=await signedIn();s.recommendations.getHomeRails=vi.fn().mockRejectedValue(new Error("offline"));s.watchlist.listWatchlist=vi.fn().mockRejectedValue(new Error("offline"));const home=await loadBrowseHome(s,"mock-profile-main");expect(home.titles).toHaveLength(5);expect(home.watchlistReady).toBe(false);expect(home.issues).toEqual(["collections","My List"]);});
 it("resolves saved titles outside the first catalog page",async()=>{const s=await signedIn();await s.watchlist.addToWatchlist("mock-profile-main",mockTitles[4].id);s.catalog.listTitles=vi.fn().mockResolvedValue({items:[mockTitles[0]],page:1,pageSize:1,total:5});s.recommendations.getHomeRails=vi.fn().mockResolvedValue([]);const home=await loadBrowseHome(s,"mock-profile-main");expect(home.titles.some(t=>t.id===mockTitles[4].id)).toBe(true);});
 it("reports unavailable metadata without claiming the saved list is empty",async()=>{const s=await signedIn();await s.watchlist.addToWatchlist("mock-profile-main",mockTitles[4].id);s.catalog.listTitles=vi.fn().mockResolvedValue({items:[],page:1,pageSize:24,total:0});s.recommendations.getHomeRails=vi.fn().mockResolvedValue([]);s.catalog.getTitle=vi.fn().mockRejectedValue(new Error("offline"));const home=await loadBrowseHome(s,"mock-profile-main");expect(home.saved).toHaveLength(1);expect(home.issues).toContain("some title information");});
 it("discards another profile's returned progress and watchlist",async()=>{const s=await signedIn();s.catalog.getContinueWatching=vi.fn().mockResolvedValue([{...progress,profileId:"other"}]);s.watchlist.listWatchlist=vi.fn().mockResolvedValue([{titleId:mockTitles[0].id,profileId:"other"}]);const home=await loadBrowseHome(s,"mock-profile-main");expect(home.progress).toEqual([]);expect(home.saved).toEqual([]);});
});
describe("development watchlist adapter",()=>{
 it("requires an authenticated verified account and valid profile/title",async()=>{const s=createMockServices();await expect(s.watchlist.listWatchlist("mock-profile-main")).rejects.toThrow();await s.auth.login({email:"demo@flurzi.example",password:"fixture"});await expect(s.watchlist.addToWatchlist("other",mockTitles[0].id)).rejects.toThrow();await expect(s.watchlist.addToWatchlist("mock-profile-main","unknown")).rejects.toThrow();});
 it("isolates profiles, prevents duplicate saves and removes the correct item",async()=>{const s=await signedIn();const title=mockTitles[0].id;await s.watchlist.addToWatchlist("mock-profile-main",title);await s.watchlist.addToWatchlist("mock-profile-main",title);await s.watchlist.addToWatchlist("mock-profile-kids",title);expect(await s.watchlist.listWatchlist("mock-profile-main")).toHaveLength(1);await s.watchlist.removeFromWatchlist("mock-profile-main",title);expect(await s.watchlist.isInWatchlist("mock-profile-main",title)).toBe(false);expect(await s.watchlist.isInWatchlist("mock-profile-kids",title)).toBe(true);});
 it("returns clones so callers cannot mutate saved state",async()=>{const s=await signedIn();await s.watchlist.addToWatchlist("mock-profile-main",mockTitles[0].id);const list=await s.watchlist.listWatchlist("mock-profile-main");list[0].titleId="changed";expect((await s.watchlist.listWatchlist("mock-profile-main"))[0].titleId).toBe(mockTitles[0].id);});
});
