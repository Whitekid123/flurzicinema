function useMatchData<T>(load:()=>Promise<T>){
 const [attempt,setAttempt]=useState(0),[state,setState]=useState<{load?:()=>Promise<T>;data?:T;error?:unknown;loading:boolean}>({loading:true});
 useEffect(()=>{let active=true;matchDeadline(Promise.resolve().then(load)).then(data=>{if(active)setState({load,data,loading:false});},error=>{if(active)setState({load,error,loading:false});});return()=>{active=false;};},[load,attempt]);
 return {data:state.load===load?state.data:undefined,error:state.load===load?state.error:undefined,loading:state.load!==load||state.loading,retry:()=>{setState({loading:true});setAttempt(n=>n+1);},replace:(data:T)=>setState({load,data,loading:false})};
}
function useMatchAction(){
 const alive=useRef(true),locked=useRef(false);const [busy,setBusy]=useState(false),[error,setError]=useState<unknown>();
 useEffect(()=>{alive.current=true;return()=>{alive.current=false;};},[]);
 return {busy,error,clear:()=>setError(undefined),run:async<T,>(work:()=>Promise<T>,done?:(data:T)=>void)=>{if(locked.current)return;locked.current=true;setBusy(true);setError(undefined);try{const data=await matchDeadline(Promise.resolve().then(work));if(alive.current)done?.(data);}catch(e){if(alive.current)setError(e);}finally{locked.current=false;if(alive.current)setBusy(false);}}};
}

function useMatchingSession(id:string){
 const service=useContext(MatchContext);const {profile}=useSession();const controller=useMemo(()=>createMatchSessionController(profile!.id,id),[profile,id]);
 const load=useMemo(()=>async()=>{const ticket=controller.begin();const [session,capabilities]=await Promise.all([service.get(id),service.capabilities()]);const accepted=controller.accept(session,ticket);if(!accepted)throw {code:'CONFLICT'};return {session:accepted,capabilities};},[service,id,controller]);const resource=useMatchData(load);const [expired,setExpired]=useState(false);
 useEffect(()=>{if(!resource.data?.session.expiresAt)return;const remaining=Date.parse(resource.data.session.expiresAt)-Date.now();const timer=setTimeout(()=>setExpired(true),Math.max(0,Math.min(remaining,2147483647)));return()=>clearTimeout(timer);},[resource.data?.session.expiresAt]);
 useEffect(()=>()=>controller.invalidate(),[controller]);
 const update=(session:MatchingSession)=>{const next=controller.accept(session);if(next&&resource.data)resource.replace({...resource.data,session:next});};
 return {...resource,error:expired?{code:'MATCH_EXPIRED'}:resource.error,loading:expired?false:resource.loading,update,service};
}

 {Boolean(error)&&<MatchFailure error={error}/>}<div className="sm-journey-actions"><button className="sm-button" disabled={busy} onClick={()=>step?setStep(step-1):onCancel()}>{step?'Back':'Cancel'}</button><div className="sm-actions">{step===0&&<button className="sm-text-button" onClick={()=>{setPrefs({...prefs,moods:[]});setStep(1);}}>Skip mood</button>}{step<2?<button className="sm-button primary" onClick={()=>setStep(step+1)}>Continue</button>:<button className="sm-button primary" disabled={busy||!consent} onClick={()=>onSubmit(prefs)}>{busy?'Sharing choices…':'Submit my choices'}</button>}</div></div></section>;
}

function useMatchWatchlist(profileId:string){
 const controller=useMemo(()=>{const service=streamingServices.watchlist;return createWatchlistController({listWatchlist:id=>matchDeadline(service.listWatchlist(id)),addToWatchlist:(id,title)=>matchDeadline(service.addToWatchlist(id,title)),removeFromWatchlist:(id,title)=>matchDeadline(service.removeFromWatchlist(id,title)),isInWatchlist:(id,title)=>matchDeadline(service.isInWatchlist(id,title))},profileId);},[profileId]);
 const state=useSyncExternalStore(controller.subscribe,controller.getSnapshot,controller.getSnapshot);useEffect(()=>controller.start(),[controller]);
 return {...state,retry:controller.retry,control:(candidate:MatchCandidate)=>({saved:!!state.items?.some(i=>i.titleId===candidate.title.id),busy:state.pending.includes(candidate.title.id),disabled:state.loading||!!state.error||!state.items,toggle:()=>void controller.toggle(candidate.title.id,candidate.title.title)})};
}
