function active(path:string,href:string){return path===href||(href!=="/browse"&&path.startsWith(href+"/"));}
export function ProfileMenu({name,avatarKey,isKids,onLogout}:{name:string;avatarKey?:string;isKids?:boolean;onLogout:()=>Promise<void>}){
 const [open,setOpen]=useState(false),[busy,setBusy]=useState(false),[error,setError]=useState("");const ref=useRef<HTMLDivElement>(null);const button=useRef<HTMLButtonElement>(null);
 useEffect(()=>{if(!open)return;const close=(event:PointerEvent)=>{if(!ref.current?.contains(event.target as Node))setOpen(false);};document.addEventListener("pointerdown",close);return()=>document.removeEventListener("pointerdown",close);},[open]);
 return <div className="cinema-profile" ref={ref} onBlur={event=>{if(!event.currentTarget.contains(event.relatedTarget))setOpen(false);}} onKeyDown={event=>{if(event.key==="Escape"){setOpen(false);button.current?.focus();}}}><button className="cinema-profile-trigger" ref={button} type="button" aria-expanded={open} aria-controls="viewer-menu" onClick={()=>setOpen(!open)}><ProfileAvatar avatarKey={avatarKey}/><span>{name}</span><span aria-hidden="true">⌄</span></button>{open&&<div className="cinema-profile-menu" id="viewer-menu"><p>Watching as <strong>{name}</strong>{isKids&&<span>Kids profile</span>}</p><Link href="/profiles">Switch profile <span aria-hidden="true">↗</span></Link><Link href="/profiles/manage">Manage profiles</Link><Link href="/account">Account settings</Link><Link href="/browse/rooms">Cinema Rooms</Link><Link href="/browse/storymatch">StoryMatch</Link><Link href="/browse/history">Viewing history</Link><button type="button" disabled={busy} onClick={async()=>{setBusy(true);setError("");try{await onLogout();}catch{setError("Could not sign out. Try again.");setBusy(false);}}}>{busy?"Signing out…":"Sign out"}</button>{error&&<p role="alert">{error}</p>}</div>}</div>;
}
